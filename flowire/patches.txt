[pre_model_sync]
# Patches in this section run before doctypes are migrated.
# Docs: https://frappeframework.com/docs/user/en/database-migrations

# Hand the script-created 'Lead Source' DocType (custom=1, module=CRM) over to the
# definition shipped in flowire/flowire/doctype/lead_source/, without dropping its
# table or its 14 rows. Before the tag patch below, so the very same migrate that
# adds the file also owns it.
flowire.patches.v1_0.own_lead_source

# Tag our schema customizations with module = "Flowire" so the fixture filters in
# hooks.py can find them.
flowire.patches.v1_0.tag_flowire_records

[post_model_sync]
# Patches in this section run after doctypes are migrated.

# Turn the old Server Scripts off in the SAME migrate that activates the Python
# doc_events in hooks.py. Deploying the code without this patch would run both
# implementations on every save.
flowire.patches.v1_0.disable_legacy_server_scripts

# Frappe's built-in "you were assigned" email off for existing users; new rows
# are handled by the Notification Settings before_insert doc event.
flowire.patches.v1_0.disable_builtin_assignment_emails

# Assignments made while the Server Scripts were off have no DocShare.
flowire.patches.v1_0.backfill_assignment_shares

# Projects that have not been touched since the costing fields were added would sit at
# zero until their next transaction. Run after the fields exist (post_model_sync runs
# after after_migrate created them).
flowire.patches.v1_0.backfill_project_costing_fields
